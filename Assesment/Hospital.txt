class Bag {

int bagId;
double price;

this.color = color;
}

public void getDetails() {
System.out.println(bagId);
System.out.println(price);
System.out.println(color);
}
}

class BagRunner{

public static void main(String[] args){

Bag b = new Bag(1, 999, "Black");

}
}
